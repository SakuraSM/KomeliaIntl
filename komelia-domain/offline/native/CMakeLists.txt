cmake_minimum_required(VERSION 3.25)
project(komelia_pdf C)
find_package(JNI REQUIRED COMPONENTS JVM)
find_library(CORE_GRAPHICS CoreGraphics REQUIRED)
find_library(CORE_FOUNDATION CoreFoundation REQUIRED)
find_library(IMAGE_IO ImageIO REQUIRED)
add_library(komelia_pdf SHARED mac_pdf.c)
target_include_directories(komelia_pdf PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(komelia_pdf ${CORE_GRAPHICS} ${CORE_FOUNDATION} ${IMAGE_IO})
